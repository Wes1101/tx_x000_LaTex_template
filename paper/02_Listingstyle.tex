%% Style of source code listings (package listings).

\renewcommand{\ttdefault}{pcr}  % monospaced font
\lstset{
    basicstyle=\ttfamily\scriptsize,
    commentstyle=\color{commentgray}\textit,
    showstringspaces=false,
    stringstyle=\color{darkgreen},
    keywordstyle=\color{blue},
    numbers=left,
    numberstyle=\tiny,
    stepnumber=1,
    numbersep=15pt,
    tabsize=2,
    fontadjust=true,
    frame=single,
    backgroundcolor=\color{boxgray},
    captionpos=t,
    linewidth=0.94\linewidth,
    xleftmargin=0.1\linewidth,
    breaklines=true,
    aboveskip=16pt,
    emph={},
    emphstyle=\bfseries,
}
