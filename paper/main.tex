    pdftitle=\Topic,
    pdfsubject=\SubTopic,
    pdfauthor=\Author,

% --- PDF-Ansicht & Lesezeichen ---
    bookmarksopen=false,
    bookmarksnumbered=true,

% --- Farbige Links aktivieren ---
    colorlinks=true,               % Text einfärben statt Boxen
    linkcolor=ObsidianPurple,             % Interne Links (Inhaltsverz., Abkürzungen, Abb.)
    citecolor=ObsidianPurple,             % Zitate (\cite)
    urlcolor=DocBlue,               % URLs und \href
    linktoc=all                    % Im Inhaltsverzeichnis Seitenzahl und Text klickbar machen
}
