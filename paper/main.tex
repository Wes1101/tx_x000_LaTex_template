\documentclass[paper=a4, fontsize=12pt, listof=totoc, bibliography=totoc, index=totoc, listof=left, parskip=full]{scrreprt}
\addtokomafont{disposition}{\rmfamily}  % serif headings

% Chapter spacing and depth of numbering / table of contents
\RedeclareSectionCommand[beforeskip=0pt, afterskip=0.5cm]{chapter}
\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{3}

\input{packages.tex}

% Wider page-number column in the lists (needed for large numbers)
\makeatletter
\renewcommand{\@pnumwidth}{32pt}
\makeatother

\input{00_Project_Data.tex}
\input{01_New_Commands.tex}

% Generate index, list of abbreviations and glossary
\makeindex
\makenomenclature
\makeglossaries

% Bibliography (biblatex, numeric style)
\addbibresource{literature/literature.bib}
\appto{\bibsetup}{\sloppy}  % allow line breaks in long URLs

% PDF metadata and link colors
\hypersetup{
    pdftitle=\Topic,
    pdfsubject=\SubTopic,
    pdfauthor=\Author,
    pdfcreator=LaTeX with hyperref,
    bookmarksopen=false,
    bookmarksnumbered=true,
    bookmarksopenlevel=1,
    pdfstartview=FitH,
    pdfpagelayout=OneColumn,
    colorlinks=true,
    linkcolor=LinkColor,
    citecolor=LinkColor,
    urlcolor=UrlColor,
    linktoc=all
}

% Layout and captions
\onehalfspacing
\setlist{noitemsep}

\addto\captionsenglish{
    \renewcommand{\figurename}{Fig.}
    \renewcommand{\tablename}{Tab.}
}
\numberwithin{table}{chapter}
\numberwithin{figure}{chapter}
\renewcommand{\thetable}{\arabic{chapter}.\arabic{table}}
\renewcommand{\thefigure}{\arabic{chapter}.\arabic{figure}}
\renewcommand{\thefootnote}{\arabic{footnote}}
\renewcommand{\multfootsep}{, }

\input{02_Listingstyle.tex}

\begin{document}
    \pagenumbering{Roman}
    \input{03_pages.tex}

    \pagenumbering{arabic}
    \input{04_chapters.tex}

    % Closing matter
    \abbr{cf.}{confer}
    \abbr{ifm}{Ingenieurgemeinschaft für Messtechnik}
    \include{pages/10_bibliography}
    \include{pages/11_Index}

    % Appendix: chapters are lettered ("Appendix A: Title"), page numbers continue.
    % Tables, figures and listings are numbered A.1, A.2, ...
    \appendix
    \renewcommand*{\chapterformat}{\appendixname~\thechapter:\enskip}
    \renewcommand{\thetable}{\Alph{chapter}.\arabic{table}}
    \renewcommand{\thefigure}{\Alph{chapter}.\arabic{figure}}
    \renewcommand{\thelstlisting}{\Alph{chapter}.\arabic{lstlisting}}
    \input{05_appendix.tex}
    \include{pages/12_AI_usage_note}
\end{document}
